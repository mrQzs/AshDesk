static int adrd_send_frame(adrd_conn *c, const frame *f)
{
    uint64_t t0 = clock_us();
    if (!c->dtls_ready)
        return ADRD_EAGAIN;
    for (size_t i = 0; i < f->slices; ++i) {
        int rc = fec_encode(&c->fec, f->slice[i]);
        if (rc < 0)
            return rc;
    }
    c->stats.enc_us = clock_us() - t0;
    return pacer_push(&c->pacer, f);
}
